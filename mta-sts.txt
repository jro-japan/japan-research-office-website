name: Scheduled Link Monitor

on:
  schedule:
    - cron: "17 2 * * 1"
  workflow_dispatch:

permissions:
  contents: read
  issues: write

jobs:
  links:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v6
      - name: Check internal and external links
        id: lychee
        uses: lycheeverse/lychee-action@v2
        with:
          args: >-
            --root-dir "${{ github.workspace }}"
            --verbose
            --no-progress
            --exclude-mail
            './**/*.html'
            './**/*.md'
            './**/*.txt'
          fail: false
          failIfEmpty: true
          format: markdown
          output: ./lychee-report.md
      - name: Create issue when broken links are found
        if: steps.lychee.outputs.exit_code != 0
        uses: peter-evans/create-issue-from-file@v6
        with:
          title: Automated broken-link report
          content-filepath: ./lychee-report.md
          labels: automated, quality, broken-link
